cmake_minimum_required(VERSION 3.20)
project(ASBridge VERSION 0.0.3 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(ascore src/cpu.c src/memory.c src/decode.c src/interpreter.c src/elf.c)
target_include_directories(ascore PUBLIC include)
target_compile_options(ascore PRIVATE -Wall -Wextra -Wpedantic)
add_executable(asrun tools/asrun.c)
target_link_libraries(asrun PRIVATE ascore)
add_executable(ascore_test tests/test_core.c)
target_link_libraries(ascore_test PRIVATE ascore)
enable_testing()
add_test(NAME ascore_v003 COMMAND ascore_test)
