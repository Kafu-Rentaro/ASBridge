cmake_minimum_required(VERSION 3.20)
project(ASBridge VERSION 0.0.4 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(ascore src/cpu.c src/memory.c src/decode.c src/interpreter.c src/elf.c src/tb.c src/jit_x86_64.c src/jit_runtime.c)
target_include_directories(ascore PUBLIC include)
target_compile_options(ascore PRIVATE -Wall -Wextra -Wpedantic)
add_executable(asrun tools/asrun.c)
target_link_libraries(asrun PRIVATE ascore)
add_executable(ascore_test tests/test_core.c)
target_link_libraries(ascore_test PRIVATE ascore)
enable_testing()
add_test(NAME ascore_v004 COMMAND ascore_test)
foreach(test_name IN ITEMS reference jit elf runtime)
  add_executable(as${test_name}_test tests/test_${test_name}.c)
  target_link_libraries(as${test_name}_test PRIVATE ascore)
  add_test(NAME as${test_name}_v004 COMMAND as${test_name}_test)
endforeach()
foreach(target IN ITEMS asrun ascore_test asreference_test asjit_test aself_test asruntime_test)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
foreach(target IN ITEMS ascore_test asreference_test asjit_test aself_test asruntime_test)
  target_compile_options(${target} PRIVATE -UNDEBUG)
endforeach()
